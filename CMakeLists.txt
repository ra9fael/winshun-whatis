cmake_minimum_required(VERSION 3.21)
project(whatis VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The native plugin requires Windows x64")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  add_compile_options(-Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
  add_link_options(-static -static-libgcc -static-libstdc++)
endif()
add_library(whatis_probe src/platform/window_probe.cpp)
target_include_directories(whatis_probe PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(whatis_probe PUBLIC user32 dwmapi)
add_executable(whatis WIN32 src/main.cpp src/ui/window.cpp)
target_link_libraries(whatis PRIVATE whatis_probe gdi32 gdiplus comctl32 shell32)
if(MSVC)
  target_link_options(whatis PRIVATE /ENTRY:mainCRTStartup)
endif()
set_target_properties(whatis PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(whatis PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}")
endforeach()
enable_testing()
add_executable(whatis_core_tests tests/core_tests.cpp)
target_include_directories(whatis_core_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME whatis_core COMMAND whatis_core_tests)
add_executable(whatis_native_tests tests/native_tests.cpp)
target_link_libraries(whatis_native_tests PRIVATE whatis_probe)
add_test(NAME whatis_native COMMAND whatis_native_tests)
add_executable(whatis_fixture tests/fixture.cpp)
target_link_libraries(whatis_fixture PRIVATE user32 gdi32 advapi32)
